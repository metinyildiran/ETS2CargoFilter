cmake_minimum_required(VERSION 3.20)
project(ETS2CargoFilter LANGUAGES CXX)

add_library(cargo_filter_core INTERFACE)
target_compile_features(cargo_filter_core INTERFACE cxx_std_20)
target_include_directories(cargo_filter_core INTERFACE src)

enable_testing()
add_executable(filter_core_test tests/filter_core_test.cpp)
target_link_libraries(filter_core_test PRIVATE cargo_filter_core)
if(MSVC)
  target_compile_options(filter_core_test PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(filter_core_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME filter_core COMMAND filter_core_test)

if(WIN32)
  add_executable(native_host_test tests/native_host_test.cpp)
  target_link_libraries(native_host_test PRIVATE cargo_filter_core)
  target_compile_options(native_host_test PRIVATE /W4 /WX /EHsc)
  add_test(NAME native_host COMMAND native_host_test)
  add_executable(native_ownership_test tests/native_ownership_test.cpp src/native_ownership.cpp)
  target_link_libraries(native_ownership_test PRIVATE cargo_filter_core)
  target_compile_options(native_ownership_test PRIVATE /W4 /WX /EHsc /utf-8)
  add_test(NAME native_ownership COMMAND native_ownership_test)
  add_executable(native_hooks_test tests/native_hooks_test.cpp src/native_hooks.cpp)
  target_link_libraries(native_hooks_test PRIVATE cargo_filter_core)
  target_compile_options(native_hooks_test PRIVATE /W4 /WX /EHsc)
  add_test(NAME native_hooks COMMAND native_hooks_test)
  add_executable(native_accept_test tests/native_accept_test.cpp src/native_accept.cpp)
  target_link_libraries(native_accept_test PRIVATE cargo_filter_core)
  target_compile_options(native_accept_test PRIVATE /W4 /WX /EHsc /utf-8)
  add_test(NAME native_accept COMMAND native_accept_test)
endif()

option(ETS2_BUILD_PLUGIN "Build the Windows x64 SDK plugin and its game adapter" OFF)
if(ETS2_BUILD_PLUGIN)
  if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The native adapter requires MSVC and a Windows x64 build.")
  endif()
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/game_adapter.cpp")
    message(FATAL_ERROR "src/game_adapter.cpp is required; no placeholder plugin is built.")
  endif()
  add_library(ETS2CargoFilter SHARED src/plugin_host.cpp src/game_adapter.cpp src/plugin_exports.def
    src/native_ownership.cpp src/native_ui.cpp src/native_rows.cpp src/native_offers.cpp src/native_hooks.cpp src/native_accept.cpp)
  target_compile_features(ETS2CargoFilter PRIVATE cxx_std_20)
  target_include_directories(ETS2CargoFilter PRIVATE src vendor/scs-sdk-1.16/include)
  target_compile_options(ETS2CargoFilter PRIVATE /W4 /WX /EHsc /utf-8)
  target_link_libraries(ETS2CargoFilter PRIVATE cargo_filter_core bcrypt)
  set_target_properties(ETS2CargoFilter PROPERTIES PREFIX "")
  add_executable(plugin_load_test tests/plugin_load_test.cpp)
  target_compile_features(plugin_load_test PRIVATE cxx_std_20)
  target_include_directories(plugin_load_test PRIVATE vendor/scs-sdk-1.16/include)
  target_compile_options(plugin_load_test PRIVATE /W4 /WX /EHsc)
  add_dependencies(plugin_load_test ETS2CargoFilter)
  add_test(NAME plugin_load COMMAND plugin_load_test $<TARGET_FILE:ETS2CargoFilter>)
endif()
